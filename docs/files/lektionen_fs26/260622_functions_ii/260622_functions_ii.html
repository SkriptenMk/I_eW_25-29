def trapezium_area(a: float, b: float, h: float) -> float:
    """Calculate the area of a trapezium.

    Args:
        a (float): The length of the first parallel side.
        b (float): The length of the second parallel side.
        h (float): The perpendicular height of the trapezium.

    Returns:
        float: The calculated area of the trapezium.
    """
    return ((a + b) / 2) * h

# test case

print(trapezium_area(2, 4, 6))

def get_cylinder_volume(r: float, h: float) -> float:
    """Calculate the volume of a cylinder.

    Args:
        r (float): The radius of the cylinder's base.
        h (float): The height of the cylinder.

    Returns:
        float: The calculated volume of the cylinder.
    """
    PI = 3.14 # Approximated value of pi to two decimal 
              # places as a local constant
    return PI * r**2 * h

def does_the_student_pass(grades: list[float]) -> bool:
    """Determine whether a student qualifies for definitive promotion.

    This function implements specific academic regulations based on a
    grading scale where 4.0 is the passing mark. A student passes if 
    both of the following conditions are met:
    1. Twice the sum of all grade deviations below 4.0 is not greater
       than the sum of all grade deviations above 4.0.
    2. No more than three grades are below 4.0.

    Args:
        grades (list[float]): A list of grades achieved in the 
            subjects relevant for promotion in the semester.

    Returns:
        bool: True if the promotion conditions are fulfilled, 
              False otherwise.
    """
    sub_fours = [g - 4 for g in grades if g < 4]
    sup_fours = [g - 4 for g in grades if g >= 4]
    aggregated_sub_fours = abs(2 * sum(sub_fours))
    aggregated_sup_fours = sum(sup_fours)
    compensation = aggregated_sub_fours <= aggregated_sup_fours
    number_sub_fours = len(sub_fours)
    return compensation and (number_sub_fours <= 3)
